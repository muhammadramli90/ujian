const API_SECRET_KEY = 'CBT-RAMLI-2026-SECURE';

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(ss, name, headers) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e2e8f0');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function doGet(e) {
  const params = e.parameter || {};
  if (params.secret !== API_SECRET_KEY) {
    return jsonResponse({ ok: false, error: 'Kunci Keamanan (API_SECRET) tidak cocok!' });
  }

  const action = params.action || 'PING';
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (action === 'PING') {
    return jsonResponse({ ok: true, message: 'Terhubung ke Google Spreadsheet: ' + ss.getName() });
  }

  if (action === 'PULL_ALL') {
    const studentSheet = getOrCreateSheet(ss, 'Data_Siswa', ['NISN', 'NAMA_LENGKAP', 'KELAS', 'PIN_LOGIN']);
    const scheduleSheet = getOrCreateSheet(ss, 'Jadwal_Ujian', ['KODE_SESI', 'MATA_PELAJARAN', 'KELAS', 'PENGAWAS', 'TANGGAL', 'JAM_MULAI', 'JAM_SELESAI', 'DURASI_MENIT', 'LINK_GFORM', 'TOKEN_PENGAWAS', 'STATUS']);
    const logSheet = getOrCreateSheet(ss, 'Log_Monitoring', ['ID_PARTICIPANT', 'KODE_SESI', 'NISN', 'NAMA_SISWA', 'KELAS', 'PERANGKAT', 'STATUS', 'PELANGGARAN', 'WAKTU_UPDATE']);

    const studentRows = studentSheet.getDataRange().getValues().slice(1);
    const students = studentRows
      .filter(r => String(r[0]).trim() !== '')
      .map(r => ({
        id: 'STD-' + String(r[0]).trim(),
        nisn: String(r[0]).trim(),
        fullName: String(r[1]).trim(),
        classroom: String(r[2]).trim(),
        pinLogin: String(r[3] || r[0]).trim()
      }));

    const scheduleRows = scheduleSheet.getDataRange().getValues().slice(1);
    const schedules = scheduleRows
      .filter(r => String(r[0]).trim() !== '')
      .map(r => ({
        id: String(r[0]).trim(),
        title: String(r[1]).trim(),
        classroom: String(r[2]).trim(),
        supervisorName: String(r[3]).trim(),
        examDate: r[4] instanceof Date ? Utilities.formatDate(r[4], Session.getScriptTimeZone(), 'yyyy-MM-dd') : String(r[4]).trim(),
        startTime: r[5] instanceof Date ? Utilities.formatDate(r[5], Session.getScriptTimeZone(), 'HH:mm') : String(r[5]).trim(),
        endTime: r[6] instanceof Date ? Utilities.formatDate(r[6], Session.getScriptTimeZone(), 'HH:mm') : String(r[6]).trim(),
        durationMinutes: Number(r[7]) || 90,
        formUrl: String(r[8]).trim(),
        unlockToken: String(r[9]).trim() || '123456',
        status: String(r[10] || 'ACTIVE').trim().toUpperCase()
      }));

    const logRows = logSheet.getDataRange().getValues().slice(1);
    const participants = logRows
      .filter(r => String(r[1]).trim() !== '' && String(r[2]).trim() !== '')
      .map(r => ({
        id: String(r[0]).trim() || ('PART-' + String(r[2]).replace(/^'/, '').trim()),
        scheduleId: String(r[1]).trim(),
        nis: String(r[2]).replace(/^'/, '').trim(),
        studentName: String(r[3]).trim(),
        classroom: String(r[4]).trim(),
        deviceLabel: String(r[5]).trim(),
        status: String(r[6]).trim() || 'IN_PROGRESS',
        violationCount: Number(r[7]) || 0,
        violationLogs: [],
        lastUpdated: Date.now()
      }));

    return jsonResponse({ ok: true, students, schedules, participants, timestamp: new Date().toISOString() });
  }

  return jsonResponse({ ok: false, error: 'Aksi GET tidak dikenal.' });
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    if (payload.secret !== API_SECRET_KEY) {
      return jsonResponse({ ok: false, error: 'Kunci Keamanan (API_SECRET) ditolak!' });
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (payload.action === 'UPSERT_PARTICIPANT' && payload.participant) {
      const p = payload.participant;
      const logSheet = getOrCreateSheet(ss, 'Log_Monitoring', ['ID_PARTICIPANT', 'KODE_SESI', 'NISN', 'NAMA_SISWA', 'KELAS', 'PERANGKAT', 'STATUS', 'PELANGGARAN', 'WAKTU_UPDATE']);
      const data = logSheet.getDataRange().getValues();
      let targetRow = -1;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][1]).trim() === String(p.scheduleId).trim() && String(data[i][2]).trim() === String(p.nis).trim()) {
          targetRow = i + 1;
          break;
        }
      }
      const rowValues = [
        p.id, p.scheduleId, "'" + p.nis, p.studentName, p.classroom, p.deviceLabel || '-',
        p.status, p.violationCount || 0, new Date().toLocaleString('id-ID')
      ];
      if (targetRow > 0) {
        logSheet.getRange(targetRow, 1, 1, 9).setValues([rowValues]);
      } else {
        logSheet.appendRow(rowValues);
      }
      return jsonResponse({ ok: true, message: 'Status peserta diperbarui di Google Sheet!' });
    }

    if (Array.isArray(payload.students)) {
      const studentSheet = getOrCreateSheet(ss, 'Data_Siswa', ['NISN', 'NAMA_LENGKAP', 'KELAS', 'PIN_LOGIN']);
      if (studentSheet.getLastRow() > 1) {
        studentSheet.getRange(2, 1, studentSheet.getLastRow() - 1, 4).clearContent();
      }
      if (payload.students.length > 0) {
        const rows = payload.students.map(s => ["'" + s.nisn, s.fullName, s.classroom, "'" + (s.pinLogin || s.nisn)]);
        studentSheet.getRange(2, 1, rows.length, 4).setValues(rows);
      }
    }

    if (Array.isArray(payload.schedules)) {
      const scheduleSheet = getOrCreateSheet(ss, 'Jadwal_Ujian', ['KODE_SESI', 'MATA_PELAJARAN', 'KELAS', 'PENGAWAS', 'TANGGAL', 'JAM_MULAI', 'JAM_SELESAI', 'DURASI_MENIT', 'LINK_GFORM', 'TOKEN_PENGAWAS', 'STATUS']);
      if (scheduleSheet.getLastRow() > 1) {
        scheduleSheet.getRange(2, 1, scheduleSheet.getLastRow() - 1, 11).clearContent();
      }
      if (payload.schedules.length > 0) {
        const rows = payload.schedules.map(j => [
          j.id, j.title, j.classroom, j.supervisorName || '-', j.examDate, j.startTime, j.endTime,
          j.durationMinutes || 90, j.formUrl, "'" + j.unlockToken, String(j.status || 'ACTIVE').trim().toUpperCase()
        ]);
        scheduleSheet.getRange(2, 1, rows.length, 11).setValues(rows);
      }
    }

    if (Array.isArray(payload.participants)) {
      const logSheet = getOrCreateSheet(ss, 'Log_Monitoring', ['ID_PARTICIPANT', 'KODE_SESI', 'NISN', 'NAMA_SISWA', 'KELAS', 'PERANGKAT', 'STATUS', 'PELANGGARAN', 'WAKTU_UPDATE']);
      if (logSheet.getLastRow() > 1) {
        logSheet.getRange(2, 1, logSheet.getLastRow() - 1, 9).clearContent();
      }
      if (payload.participants.length > 0) {
        const rows = payload.participants.map(p => [
          p.id, p.scheduleId, "'" + p.nis, p.studentName, p.classroom, p.deviceLabel || '-',
          p.status, p.violationCount || 0, new Date(p.lastUpdated || Date.now()).toLocaleString('id-ID')
        ]);
        logSheet.getRange(2, 1, rows.length, 9).setValues(rows);
      }
    }

    return jsonResponse({ ok: true, message: 'Data berhasil disinkronkan ke Google Spreadsheet!' });
  } catch (err) {
    return jsonResponse({ ok: false, error: err.toString() });
  }
}
